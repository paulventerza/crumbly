import React, { useState, useEffect } from 'react';
import { Clock, Plus, ChevronRight, Save, ArrowLeft, ChefHat, Timer, Trash2, ListChecks, Info, Bell, BellOff, CheckCircle2 } from 'lucide-react';

// --- Utility Functions for Time Math ---

const formatTime = (date) => {
  return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', hour12: false });
};

const subtractMinutes = (date, minutes) => {
  return new Date(date.getTime() - minutes * 60000);
};

const addMinutes = (date, minutes) => {
  return new Date(date.getTime() + minutes * 60000);
};

// --- Mock Data ---
const DEFAULT_RECIPES = [
  {
    id: '1',
    name: 'Classic Sourdough',
    ingredients: ['500g Bread Flour', '350g Water', '100g Starter', '10g Salt'],
    steps: [
      { id: 's1', name: 'Autolyse', instruction: 'Mix flour and water until no dry bits remain.', activeTime: 10, waitTime: 60 },
      { id: 's2', name: 'Add Levain', instruction: 'Pinch in the starter and salt.', activeTime: 10, waitTime: 30 },
      { id: 's3', name: 'Fold 1', instruction: 'Perform a set of stretch and folds.', activeTime: 5, waitTime: 30 },
      { id: 's4', name: 'Bulk Ferment', instruction: 'Leave in a warm spot until doubled.', activeTime: 0, waitTime: 240 },
      { id: 's5', name: 'Bake', instruction: 'Preheat dutch oven to 230°C.', activeTime: 45, waitTime: 0 },
    ]
  }
];

// --- Components ---

const Button = ({ children, onClick, variant = 'primary', className = '', disabled = false }) => {
  const baseStyle = "px-4 py-3 rounded-xl font-medium transition-all flex items-center justify-center gap-2 active:scale-95";
  const variants = {
    primary: "bg-orange-600 text-white shadow-lg shadow-orange-200 hover:bg-orange-700 disabled:opacity-50 disabled:cursor-not-allowed",
    secondary: "bg-stone-200 text-stone-700 hover:bg-stone-300",
    outline: "border-2 border-orange-600 text-orange-600 hover:bg-orange-50",
    danger: "bg-red-100 text-red-600 hover:bg-red-200",
    ghost: "text-stone-400 hover:text-red-500"
  };
  
  return (
    <button onClick={onClick} disabled={disabled} className={`${baseStyle} ${variants[variant]} ${className}`}>
      {children}
    </button>
  );
};

const Input = ({ label, type = "text", value, onChange, placeholder, min }) => (
  <div className="flex flex-col gap-1 mb-4">
    <label className="text-stone-600 text-sm font-semibold uppercase tracking-wider">{label}</label>
    <input 
      type={type} 
      value={value} 
      onChange={onChange} 
      min={min}
      placeholder={placeholder}
      className="p-3 bg-white border border-stone-300 rounded-lg focus:ring-2 focus:ring-orange-500 focus:outline-none text-stone-800"
    />
  </div>
);

// --- Main App Component ---

export default function App() {
  const [view, setView] = useState('home');
  const [recipes, setRecipes] = useState(DEFAULT_RECIPES);
  const [selectedRecipe, setSelectedRecipe] = useState(null);
  const [isDeleteMode, setIsDeleteMode] = useState(false);
  const [currentTime, setCurrentTime] = useState(new Date());
  const [notificationsEnabled, setNotificationsEnabled] = useState(false);
  
  // Creation State
  const [newRecipeName, setNewRecipeName] = useState('');
  const [newIngredients, setNewIngredients] = useState([]);
  const [currentIngredient, setCurrentIngredient] = useState('');
  const [newSteps, setNewSteps] = useState([]);
  
  // New Step Form State
  const [currentStepName, setCurrentStepName] = useState('');
  const [currentInstruction, setCurrentInstruction] = useState('');
  const [currentActiveTime, setCurrentActiveTime] = useState('');
  const [currentWaitTime, setCurrentWaitTime] = useState('');

  // Scheduling State
  const [eatTime, setEatTime] = useState('19:00');
  const [schedule, setSchedule] = useState([]);

  // Tick current time for highlighting
  useEffect(() => {
    const timer = setInterval(() => setCurrentTime(new Date()), 30000); // Update every 30s
    return () => clearInterval(timer);
  }, []);

  const requestNotificationPermission = async () => {
    if (!("Notification" in window)) return;
    const permission = await Notification.requestPermission();
    if (permission === "granted") {
      setNotificationsEnabled(true);
    }
  };

  const handleAddIngredient = () => {
    if (currentIngredient.trim()) {
      setNewIngredients([...newIngredients, currentIngredient.trim()]);
      setCurrentIngredient('');
    }
  };

  const handleCreateStep = () => {
    if (!currentStepName || !currentActiveTime) return;
    const step = {
      id: Date.now().toString(),
      name: currentStepName,
      instruction: currentInstruction,
      activeTime: parseInt(currentActiveTime) || 0,
      waitTime: parseInt(currentWaitTime) || 0
    };
    setNewSteps([...newSteps, step]);
    setCurrentStepName('');
    setCurrentInstruction('');
    setCurrentActiveTime('');
    setCurrentWaitTime('');
  };

  const handleSaveRecipe = () => {
    if (!newRecipeName || newSteps.length === 0) return;
    const recipe = {
      id: Date.now().toString(),
      name: newRecipeName,
      ingredients: newIngredients,
      steps: newSteps
    };
    setRecipes([...recipes, recipe]);
    setNewRecipeName('');
    setNewIngredients([]);
    setNewSteps([]);
    setView('home');
  };

  const deleteRecipe = (e, id) => {
    e.stopPropagation();
    setRecipes(recipes.filter(r => r.id !== id));
  };

  const calculateSchedule = (recipe, targetTimeStr) => {
    const [targetHours, targetMinutes] = targetTimeStr.split(':');
    let finishDate = new Date();
    finishDate.setHours(parseInt(targetHours), parseInt(targetMinutes), 0, 0);

    if (finishDate < new Date()) {
        finishDate.setDate(finishDate.getDate() + 1);
    }

    let runningTime = new Date(finishDate);
    const calculatedSteps = [];

    for (let i = recipe.steps.length - 1; i >= 0; i--) {
      const step = recipe.steps[i];
      runningTime = subtractMinutes(runningTime, step.waitTime);
      const activeFinishTime = new Date(runningTime);
      runningTime = subtractMinutes(runningTime, step.activeTime);
      const startTime = new Date(runningTime);

      calculatedSteps.unshift({
        ...step,
        startTime: startTime,
        endTime: activeFinishTime,
        nextStepTime: addMinutes(activeFinishTime, step.waitTime)
      });
    }
    setSchedule(calculatedSteps);
    
    // Set up notifications if enabled
    if (notificationsEnabled) {
      calculatedSteps.forEach(step => {
        const timeout = step.startTime.getTime() - new Date().getTime();
        if (timeout > 0) {
          setTimeout(() => {
            new Notification("Crumbly Time!", {
              body: `Start: ${step.name}`,
              icon: 'https://cdn-icons-png.flaticon.com/512/992/992747.png'
            });
          }, timeout);
        }
      });
    }
  };

  useEffect(() => {
    if (view === 'schedule' && selectedRecipe) {
      calculateSchedule(selectedRecipe, eatTime);
    }
  }, [eatTime, view, selectedRecipe, notificationsEnabled]);

  // --- Views ---

  if (view === 'home') {
    return (
      <div className="min-h-screen bg-stone-50 text-stone-800 font-sans p-4 sm:p-6 md:max-w-md mx-auto relative">
        <header className="mb-8 pt-4 flex justify-between items-start">
          <div>
            <div className="flex items-center gap-3 mb-2">
              <div className="bg-orange-600 p-2 rounded-lg shadow-md">
                <ChefHat className="text-white w-6 h-6" />
              </div>
              <h1 className="text-2xl font-bold text-stone-900 tracking-tight">Crumbly</h1>
            </div>
            <p className="text-stone-500 italic">Rise to the occasion</p>
          </div>
          <button 
            onClick={() => setIsDeleteMode(!isDeleteMode)}
            className={`p-2 rounded-lg transition-colors ${isDeleteMode ? 'bg-red-500 text-white' : 'bg-stone-200 text-stone-500'}`}
          >
            <Trash2 className="w-5 h-5" />
          </button>
        </header>

        <div className="grid gap-4">
          {recipes.map(recipe => (
            <div 
              key={recipe.id} 
              onClick={() => { if(!isDeleteMode) { setSelectedRecipe(recipe); setView('schedule'); } }}
              className={`bg-white p-5 rounded-2xl shadow-sm border transition-all relative overflow-hidden ${isDeleteMode ? 'border-red-200 animate-pulse' : 'border-stone-200 cursor-pointer hover:border-orange-300 active:scale-[0.98] group'}`}
            >
              <div className="flex justify-between items-center">
                <div>
                  <h3 className="font-bold text-lg text-stone-800 group-hover:text-orange-700">{recipe.name}</h3>
                  <p className="text-stone-500 text-sm mt-1">{recipe.steps.length} Steps • {recipe.ingredients?.length || 0} Ingredients</p>
                </div>
                {isDeleteMode ? (
                  <button onClick={(e) => deleteRecipe(e, recipe.id)} className="bg-red-500 text-white p-3 rounded-full shadow-lg z-10">
                    <Trash2 className="w-5 h-5" />
                  </button>
                ) : (
                  <div className="bg-orange-50 p-2 rounded-full group-hover:bg-orange-100">
                    <ChevronRight className="w-5 h-5 text-orange-600" />
                  </div>
                )}
              </div>
            </div>
          ))}

          <button 
            onClick={() => setView('create')}
            className="mt-4 border-2 border-dashed border-stone-300 rounded-2xl p-4 flex flex-col items-center justify-center gap-2 text-stone-500 hover:border-orange-400 hover:text-orange-600 hover:bg-orange-50 transition-all h-32"
          >
            <Plus className="w-8 h-8" />
            <span className="font-medium">Create New Recipe</span>
          </button>
        </div>
      </div>
    );
  }

  if (view === 'create') {
    return (
      <div className="min-h-screen bg-stone-50 text-stone-800 font-sans p-4 sm:p-6 md:max-w-md mx-auto pb-24">
        <header className="flex items-center gap-4 mb-6 pt-2">
          <button onClick={() => setView('home')} className="p-2 hover:bg-stone-200 rounded-full">
            <ArrowLeft className="w-6 h-6 text-stone-600" />
          </button>
          <h2 className="text-xl font-bold">New Recipe</h2>
        </header>

        <div className="space-y-6">
          <div className="bg-white p-5 rounded-2xl shadow-sm border border-stone-100">
            <Input label="Recipe Name" placeholder="e.g., Rustic White Loaf" value={newRecipeName} onChange={(e) => setNewRecipeName(e.target.value)} />
          </div>

          <div className="bg-white p-5 rounded-2xl shadow-sm border border-stone-100">
            <h3 className="font-bold text-stone-800 mb-4 flex items-center gap-2"><ListChecks className="w-4 h-4 text-orange-600" /> Ingredients</h3>
            <div className="flex gap-2 mb-4">
              <input 
                className="flex-1 p-2 border border-stone-300 rounded-lg text-sm focus:ring-2 focus:ring-orange-500 focus:outline-none"
                placeholder="Add ingredient..."
                value={currentIngredient}
                onChange={(e) => setCurrentIngredient(e.target.value)}
                onKeyPress={(e) => e.key === 'Enter' && handleAddIngredient()}
              />
              <button onClick={handleAddIngredient} className="bg-stone-800 text-white p-2 rounded-lg"><Plus className="w-5 h-5"/></button>
            </div>
            <ul className="space-y-2">
              {newIngredients.map((ing, i) => (
                <li key={i} className="text-sm bg-stone-50 p-2 rounded border border-stone-100 flex justify-between group">
                  {ing}
                  <button onClick={() => setNewIngredients(newIngredients.filter((_, idx) => idx !== i))} className="text-stone-300 hover:text-red-500"><Trash2 className="w-4 h-4"/></button>
                </li>
              ))}
            </ul>
          </div>

          <div className="bg-white p-5 rounded-2xl shadow-sm border border-stone-100">
            <h3 className="font-bold text-stone-800 mb-4">Step {newSteps.length + 1}</h3>
            <Input label="Step Title" placeholder="e.g., Mix Dough" value={currentStepName} onChange={(e) => setCurrentStepName(e.target.value)} />
            <Input label="Instructions" placeholder="Details of what to do..." value={currentInstruction} onChange={(e) => setCurrentInstruction(e.target.value)} />
            
            <div className="grid grid-cols-2 gap-4">
              <Input label="Work (min)" type="number" value={currentActiveTime} onChange={(e) => setCurrentActiveTime(e.target.value)} />
              <Input label="Wait After (min)" type="number" value={currentWaitTime} onChange={(e) => setCurrentWaitTime(e.target.value)} />
            </div>

            <Button variant="secondary" className="w-full" onClick={handleCreateStep} disabled={!currentStepName || !currentActiveTime}>
              <Plus className="w-4 h-4" /> Add Step
            </Button>
          </div>

          {newSteps.length > 0 && (
            <div className="space-y-4">
               <div className="space-y-2">
                {newSteps.map((step, idx) => (
                    <div key={idx} className="bg-white p-3 rounded-lg border border-stone-200 flex justify-between items-center text-sm">
                      <span className="font-medium">{idx + 1}. {step.name}</span>
                      <span className="text-stone-400">{step.activeTime}m / {step.waitTime}m</span>
                    </div>
                  ))}
               </div>
               <Button className="w-full fixed bottom-6 left-1/2 -translate-x-1/2 max-w-[calc(100%-3rem)] md:max-w-[350px]" onClick={handleSaveRecipe}>
                  <Save className="w-5 h-5" /> Save Recipe
               </Button>
            </div>
          )}
        </div>
      </div>
    );
  }

  if (view === 'schedule' && selectedRecipe) {
    const firstStepStart = schedule.length > 0 ? schedule[0].startTime : new Date();

    return (
      <div className="min-h-screen bg-stone-50 text-stone-800 font-sans md:max-w-md mx-auto flex flex-col">
        <div className="bg-white shadow-sm p-4 sticky top-0 z-10">
            <div className="flex items-center justify-between mb-4">
                <button onClick={() => setView('home')} className="p-2 -ml-2 hover:bg-stone-100 rounded-full">
                    <ArrowLeft className="w-6 h-6 text-stone-600" />
                </button>
                <h2 className="font-bold text-lg truncate flex-1 text-center">{selectedRecipe.name}</h2>
                <button 
                  onClick={requestNotificationPermission}
                  className={`p-2 rounded-full ${notificationsEnabled ? 'text-orange-600 bg-orange-50' : 'text-stone-300 hover:bg-stone-50'}`}
                >
                  {notificationsEnabled ? <Bell className="w-5 h-5" /> : <BellOff className="w-5 h-5" />}
                </button>
            </div>

            <div className="bg-orange-50 rounded-xl p-4 flex items-center justify-between border border-orange-100">
                <div className="flex flex-col">
                    <label className="text-orange-800 text-[10px] font-bold uppercase mb-1 tracking-wider">Ready to Eat</label>
                    <input type="time" value={eatTime} onChange={(e) => setEatTime(e.target.value)} className="bg-transparent text-2xl font-bold text-orange-900 focus:outline-none" />
                </div>
                <div className="text-right">
                    <p className="text-[10px] text-orange-700 font-bold uppercase tracking-wider mb-1">Start Baking</p>
                    <p className="text-2xl font-bold text-orange-800">{formatTime(firstStepStart)}</p>
                </div>
            </div>

            {selectedRecipe.ingredients?.length > 0 && (
              <div className="mt-4 pt-4 border-t border-stone-100">
                <div className="flex items-center gap-2 mb-2">
                  <ListChecks className="w-4 h-4 text-stone-400" />
                  <span className="text-xs font-bold uppercase text-stone-500 tracking-widest">Ingredients</span>
                </div>
                <div className="flex flex-wrap gap-2">
                  {selectedRecipe.ingredients.map((ing, i) => (
                    <span key={i} className="text-xs bg-stone-100 text-stone-600 px-2 py-1 rounded-md border border-stone-200">{ing}</span>
                  ))}
                </div>
              </div>
            )}
        </div>

        <div className="flex-1 p-4 overflow-y-auto">
            <div className="relative pl-6 space-y-6 pb-20 before:absolute before:left-2.5 before:top-2 before:bottom-0 before:w-0.5 before:bg-stone-200">
                {schedule.map((step, idx) => {
                    const isLast = idx === schedule.length - 1;
                    const isCurrent = currentTime >= step.startTime && currentTime < (isLast ? addMinutes(step.startTime, step.activeTime + 30) : schedule[idx+1].startTime);
                    const isPast = currentTime > (isLast ? addMinutes(step.startTime, step.activeTime + 30) : schedule[idx+1].startTime);

                    return (
                        <div key={idx} className={`relative transition-all duration-500 ${isCurrent ? 'scale-105 z-20' : ''}`}>
                            <div className={`absolute -left-[27px] top-1 w-5 h-5 rounded-full border-4 z-10 transition-colors ${isPast ? 'bg-orange-600 border-orange-200' : isCurrent ? 'bg-white border-orange-500 animate-pulse' : 'bg-white border-stone-200'}`}>
                                {isPast && <CheckCircle2 className="w-3 h-3 text-white absolute inset-0 m-auto" />}
                            </div>
                            
                            <div className={`p-4 rounded-xl border shadow-sm transition-all ${isCurrent ? 'bg-white border-orange-400 ring-4 ring-orange-50 shadow-orange-100' : isPast ? 'bg-stone-50 border-stone-100 opacity-60' : 'bg-white border-stone-200'}`}>
                                <div className="flex justify-between items-start mb-1">
                                    <div className="flex flex-col">
                                      {isCurrent && <span className="text-[10px] font-black text-orange-600 tracking-tighter mb-0.5">CURRENTLY</span>}
                                      <h4 className={`font-bold ${isCurrent ? 'text-orange-900' : 'text-stone-900'}`}>{step.name}</h4>
                                    </div>
                                    <div className={`px-2 py-1 rounded text-xs font-mono font-bold ${isCurrent ? 'bg-orange-600 text-white' : 'bg-stone-800 text-white'}`}>
                                        {formatTime(step.startTime)}
                                    </div>
                                </div>
                                
                                {step.instruction && (
                                  <p className="text-sm text-stone-600 mb-3 leading-relaxed flex items-start gap-1.5 bg-stone-50 p-2 rounded-lg italic">
                                    <Info className="w-3.5 h-3.5 mt-0.5 text-stone-400 shrink-0" />
                                    {step.instruction}
                                  </p>
                                )}

                                <div className="flex gap-4 text-[11px] font-bold uppercase tracking-wide text-stone-400">
                                    <div className="flex items-center gap-1">
                                        <Timer className="w-3.5 h-3.5" />
                                        <span>{step.activeTime}m work</span>
                                    </div>
                                    {!isLast && step.waitTime > 0 && (
                                        <div className={`flex items-center gap-1 ${isCurrent ? 'text-orange-600 font-black' : 'text-stone-400'}`}>
                                            <Clock className="w-3.5 h-3.5" />
                                            <span>Wait {step.waitTime}m</span>
                                        </div>
                                    )}
                                </div>
                            </div>
                        </div>
                    );
                })}
                 
                 <div className="relative">
                    <div className={`absolute -left-[27px] top-1 w-5 h-5 rounded-full border-2 z-10 shadow-sm ${currentTime >= new Date(schedule[schedule.length-1]?.startTime).getTime() ? 'bg-orange-600 border-orange-100' : 'bg-stone-200 border-white'}`}></div>
                    <div className={`${currentTime >= new Date(schedule[schedule.length-1]?.startTime).getTime() ? 'bg-orange-600' : 'bg-stone-300'} text-white p-3 rounded-xl shadow-md flex justify-between items-center transition-colors`}>
                        <span className="font-bold uppercase tracking-widest text-sm">Eat!</span>
                        <span className="font-mono font-bold text-lg">{eatTime}</span>
                    </div>
                 </div>
            </div>
        </div>
      </div>
    );
  }

  return null;
}
